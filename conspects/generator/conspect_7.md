# Ячейка 6: Анализ скейлинга
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from scipy.stats import linregress

# Загружаем данные (если файлы в папке)
data60 = pd.read_csv('xi_n60.csv')
data80 = pd.read_csv('xi_n80.csv')
data100 = pd.read_csv('xi_n100.csv')

fc = 0.5

def prepare_data(df, n):
    df = df[abs(df['f'] - fc) < 0.1].copy()
    df['df'] = np.abs(df['f'] - fc)
    df['log_df'] = np.log(df['df'])
    df['log_xi'] = np.log(df['xi'])
    # отбрасываем нулевые или очень малые ξ
    df = df[df['xi'] > 0.1]
    df['xi_err_rel'] = df['xi_err'] / df['xi']
    return df

df60 = prepare_data(data60, 60)
df80 = prepare_data(data80, 80)
df100 = prepare_data(data100, 100)

# Регрессия и вывод
print("Оценка показателя ν:")
for n, df in zip([60,80,100], [df60, df80, df100]):
    if len(df) < 3:
        print(f"n={n}: недостаточно данных")
        continue
    slope, intercept, r_value, p_value, stderr_slope = linregress(df['log_df'], df['log_xi'])
    nu = slope
    nu_err = stderr_slope
    print(f"n={n}: ν = {nu:.3f} ± {nu_err:.3f}, R² = {r_value**2:.3f}")

# Построение графика
plt.figure(figsize=(8,6))
colors = {'60':'blue', '80':'green', '100':'red'}
for n, df, color in zip([60,80,100], [df60, df80, df100], ['blue','green','red']):
    if len(df) == 0:
        continue
    plt.errorbar(df['df'], df['xi'], yerr=df['xi_err'], fmt='o', color=color, label=f'n={n}', capsize=3)
    # линия регрессии
    slope, intercept, _, _, _ = linregress(np.log(df['df']), np.log(df['xi']))
    x_fit = np.linspace(df['df'].min(), df['df'].max(), 50)
    y_fit = np.exp(intercept) * x_fit**slope
    plt.plot(x_fit, y_fit, '--', color=color, alpha=0.7)

plt.xscale('log')
plt.yscale('log')
plt.xlabel('|f - f_c|')
plt.ylabel('ξ')
plt.title('Скейлинг корреляционной длины')
plt.legend()
plt.grid(True, which='both', alpha=0.3)
plt.tight_layout()
plt.savefig('scaling_n60_80_100.png', dpi=150)
plt.show()
