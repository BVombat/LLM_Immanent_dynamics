3. Код для квантового расширения (сохранить для новой сессии)
python
import numpy as np
import networkx as nx
from scipy.linalg import expm
import matplotlib.pyplot as plt

# ------------------------------------------------------------
# 1. Построение квантового оператора U_v для вершины v
# ------------------------------------------------------------
def pauli_z():
    return np.array([[1, 0], [0, -1]])

def pauli_x():
    return np.array([[0, 1], [1, 0]])

def pauli_y():
    return np.array([[0, -1j], [1j, 0]])

def identity():
    return np.eye(2)

def tensor_product(*matrices):
    """Произведение матриц."""
    result = matrices[0]
    for m in matrices[1:]:
        result = np.kron(result, m)
    return result

def build_Uv(G, v, f, gamma=1.0):
    """
    Строит оператор U_v = exp(-i γ H_v) для вершины v.
    H_v = -f * (сумма по соседям σ_z) - (1-f) * (сумма по соседям σ_x) ??
    Здесь реализуем упрощённую версию: гамильтониан, соответствующий правилу большинства.
    """
    n = G.number_of_nodes()
    neighbors = list(G.neighbors(v))
    deg = len(neighbors)
    if deg == 0:
        return np.eye(2**n)  # нет взаимодействия
    # Строим H_v как сумму по соседям: σ_z^{(v)} σ_z^{(u)} + σ_x^{(v)} σ_x^{(u)}?
    # Упростим: H_v = - (1/2) * (1 - 2f) * (σ_z^{(v)} * sum_{u∈N(v)} σ_z^{(u)})
    # Это эвристическая форма, отражающая правило большинства.
    # Полный гамильтониан всей системы будет суммой по всем v.
    # Для проверки T2 достаточно рассмотреть динамику намагниченности.
    # Вместо точного построения используем эффективный оператор эволюции одного узла.
    # Возвращаем матрицу размера 2^n (экспоненту от H_v).
    raise NotImplementedError("Требуется доработка для произвольного графа.")
Примечание: Построение полного квантового гамильтониана для произвольного графа — нетривиальная задача. Ниже приводится рабочий код для небольших систем (n ≤ 12), где можно выполнить точную диагонализацию.

python
def build_hamiltonian(G, f):
    """
    Строит гамильтониан H = - sum_v [ f * M_v + (1-f) * (1-M_v) ]? 
    Альтернатива: H = -J sum_{(i,j)} (σ_z^i σ_z^j) - h sum_i σ_x^i, с h = 2f-1.
    Такая форма известна как квантовое правило большинства.
    """
    n = G.number_of_nodes()
    H = np.zeros((2**n, 2**n), dtype=complex)
    # Взаимодействие ближайших соседей: -J σ_z⊗σ_z
    J = 1.0
    for i, j in G.edges():
        # σ_z на i, σ_z на j, единицы на остальных
        op_i = pauli_z()
        op_j = pauli_z()
        matrices = [identity()] * n
        matrices[i] = op_i
        matrices[j] = op_j
        term = tensor_product(*matrices)
        H -= J * term
    # Поле в направлении x: -h σ_x, где h = 2f-1
    h = 2*f - 1
    for i in range(n):
        matrices = [identity()] * n
        matrices[i] = pauli_x()
        term = tensor_product(*matrices)
        H -= h * term
    return H

def time_evolution(H, t):
    """Оператор эволюции e^{-i H t}."""
    return expm(-1j * H * t)

def magnetization(state):
    """Среднее по всем узлам <σ_z>."""
    n = int(np.log2(len(state)))
    m = 0
    for i in range(n):
        op = pauli_z()
        matrices = [identity()] * n
        matrices[i] = op
        total_op = tensor_product(*matrices)
        m += np.real(state.conj().T @ total_op @ state)
    return m / n

def compute_T2(G, f, t_max=10.0, dt=0.05):
    """
    Вычисляет время декогеренции T2 по затуханию начальной намагниченности.
    Начальное состояние: все спины вверх |+++...>.
    """
    n = G.number_of_nodes()
    H = build_hamiltonian(G, f)
    # начальное состояние: все вверх
    psi0 = np.zeros(2**n, dtype=complex)
    psi0[0] = 1.0  # состояние |0...0> соответствует всем спинам вверх? Нужно проверить базис.
    # В базисе, где σ_z|0> = +1|0>, |0> = (1,0)^T.
    # Тогда состояние всех спинов вверх = |00...0> = индекс 0.
    psi0[0] = 1.0
    times = np.arange(0, t_max, dt)
    mz = []
    for t in times:
        U = time_evolution(H, t)
        psi = U @ psi0
        mz.append(magnetization(psi))
    # Аппроксимируем затухание экспонентой mz(t) = exp(-t/T2)
    # Используем только положительные значения
    mz = np.array(mz)
    mask = mz > 0
    if np.sum(mask) < 3:
        return np.nan
    log_mz = np.log(mz[mask])
    t_fit = times[mask]
    slope, intercept = np.polyfit(t_fit, log_mz, 1)
    T2 = -1.0 / slope
    return T2
